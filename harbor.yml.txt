---
- name: Install Harbor dependencies
  hosts: webservers
  become: yes

  tasks:
    - name: Install Docker
      apt:
        name:
          - docker.io
          - docker-compose-plugin
        state: present
        update_cache: yes

    - name: Start Docker
      service:
        name: docker
        state: started
        enabled: yes

    - name: Create Harbor directory
      file:
        path: /opt/harbor
        state: directory
        mode: '0755'

    - name: Download Harbor installer
      get_url:
        url: https://github.com/goharbor/harbor/releases/download/v2.13.2/harbor-online-installer-v2.13.2.tgz
        dest: /tmp/harbor.tgz

    - name: Extract Harbor
      unarchive:
        src: /tmp/harbor.tgz
        dest: /opt/
        remote_src: yes
