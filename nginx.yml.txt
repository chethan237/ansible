---
- name: Install Nginx
  hosts: webservers
  become: yes

  tasks:
    - name: Update apt cache
      apt:
        update_cache: yes

    - name: Install Nginx
      apt:
        name: nginx
        state: present

    - name: Start Nginx
      service:
        name: nginx
        state: started
        enabled: yes

    - name: Check Nginx status
      command: nginx -t
      register: nginx_test

    - name: Display Nginx status
      debug:
        var: nginx_test.stdout
